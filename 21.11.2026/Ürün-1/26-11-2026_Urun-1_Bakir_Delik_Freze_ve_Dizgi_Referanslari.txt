Yeniden çalıştırma eki
Birinci kaynağın sabitlenmiş sürümündeki gerber_check.py dosyasını ve aşağıdaki replay.py içeriğini boş bir klasöre yerleştirin. Python, NumPy, SciPy ve Pillow kurulu ortamda “python replay.py” dört paketi üretir, denetimi çalıştırır ve beklenen bulgu sayılarını ve çıkış kodunu karşılaştırır. Bu koşuda NumPy 2.3.5, SciPy 1.17.0 ve Pillow 13.0.0 kullanıldı. gerber_check.py Git blob kimliği 9a80a2a25e37abf023b38893a24767e4ffd94640.

--- replay.py ---
from pathlib import Path
import re
import subprocess
import sys

root = Path(__file__).resolve().parent
def layer(side, commands):
    return "\n".join(["%FSLAX24Y24*%", "%MOMM*%", f"%TF.FileFunction,Copper,{side}*%", "%ADD10C,0.600*%", "D10*", *commands, "M02*"]) + "\n"

def point(x, y, d):
    return f"X{x*10000:06d}Y{y*10000:06d}D{d}*"

def line(x1, y1, x2, y2):
    return [point(x1,y1,"02"), point(x2,y2,"01")]

def pad(code, net, ref, x, y, access):
    return f"{code}{net:<14}   {ref:<6}-{'1':<4} {'D0300P' if code=='317' else '      '}A{access:02d}X{x*1000:+07d}Y{y*1000:+07d}X0600Y0600"

ipc = ["P  UNITS CUST 1", pad("327","A","J1",5,5,1), pad("317","A","TP1",10,5,0), pad("327","A","J2",15,5,2), pad("327","B","J3",5,8,1), pad("327","B","J4",15,8,1), "999"]
(root / "board.ipc").write_text("\n".join(ipc) + "\n")

cases = {"clean": (0,0,0,0), "open": (0,1,0,0), "short": (1,0,0,0), "missing_drill": (0,1,0,1)}
for name, expected in cases.items():
    folder = root / name
    folder.mkdir(exist_ok=True)
    top = line(5,5,10,5) + line(5,8,15,8)
    if name == "short":
        top += line(8,5,8,8)
    bot = line(10,5,15,5) if name != "open" else line(10,5,12,5) + line(13,5,15,5)
    (folder / "top.gtl").write_text(layer("L1,Top", top))
    (folder / "bottom.gbl").write_text(layer("L2,Bot", bot))
    drill = ["M48","METRIC,TZ","T01C0.300","%","T01"]
    if name != "missing_drill":
        drill.append("X10.000Y5.000")
    (folder / "PTH.drl").write_text("\n".join(drill + ["M30"])+"\n")
    run = subprocess.run([sys.executable, str(root / "gerber_check.py"), "netcheck", str(folder), "--ipc", str(root / "board.ipc"), "--no-auto"],capture_output=True,text=True)
    print(name, run.returncode, run.stdout, run.stderr)
    summary = re.search(r"SUMMARY shorts (\d+), opens (\d+), pads without copper (\d+), undrilled (\d+)",run.stdout)
    assert summary and tuple(map(int,summary.groups())) == expected and run.returncode == int(any(expected))
